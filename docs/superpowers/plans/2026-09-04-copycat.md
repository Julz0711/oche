# Copycat Game Module Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Copycat as a third playable `GameModule` — a "matching chase" darts variant where players throw 3-dart visits trying to match the previous visit's total, the point-winner leads the next point, and the deciding point requires a double/inner-bull finish.

**Architecture:** Follows the exact `GameModule` file structure and conventions X01 and Cricket already establish: a pure state-transition engine with snapshot-based undo, a settings schema, a `computeStats`/`matchStats` split, a `ScoreboardSlot` component (no `HistoryPanel` override — reuses the generic one), wired into the shared registry. No new infrastructure — this plan is entirely additive, reusing the `raceProgress`/`matchInfo`/`HistoryPanel`-override mechanisms already built.

**Tech Stack:** TypeScript, Zod (settings schema), Vitest (engine tests), React (ScoreboardSlot).

**Spec:** [2026-09-04-copycat-design.md](../specs/2026-09-04-copycat-design.md)

## Global Constraints

- Never trust an incoming `Throw.points` field at the point of play — every dart's points must be recomputed from `sector`+`ring` inside the engine (same discipline X01/Cricket already follow).
- `isDoubleFinish`-equivalent logic must be defined locally in Copycat's own engine file — do not import X01's private `isDoubleRing` helper (it isn't exported, and game modules don't import each other's internals; X01 and Cricket don't).
- `firstThrow` settings enum (`player1`/`random`/`bulloff`/`alternate`) must match X01's actual implementation depth exactly: only `'random'` behaves differently (picks a random starting index); the other three all default to index 0. Do not invent real bull-off/alternate logic X01 itself doesn't have.
- `playerRange: [2, 8]`, matching X01/Cricket.
- Icon: `Repeat` from `lucide-react` — not `Copy` (already used elsewhere in this app for a different meaning).
- No `HistoryPanel` override for this version — Copycat relies on the generic per-game dart-by-dart history already used by X01.
- `computeStats` (operates on the generic persisted `Visit[]` shape) must honestly return `null` for any stat it cannot reconstruct from that flat shape (no `targetBefore`/point-win context available there) — do not fake a value. `matchStats` (operates on live in-memory state) computes everything fully.

```ts
// lib/games/copycat/engine.ts
import { Throw, PlayerId } from '@/lib/types';
import { CopycatSettings } from './schema';

/** Derives a dart's real point value from its sector+ring — never trust an incoming Throw's
 *  `points` field; always recompute it here (same discipline every engine in this project
 *  follows, established during M4's interface-discipline fix). */
function computeDartPoints(sector: Throw['sector'], ring: Throw['ring']): number {
  if (ring === 'M') return 0;
  if (ring === 'OB') return 25;
  if (ring === 'IB') return 50;
  return sector * (ring === 'D' ? 2 : ring === 'T' ? 3 : 1);
}

/** A double or the inner bull — the same "valid finish" concept X01 uses for checkOut:
 *  'double'. Reimplemented as its own one-liner here since X01's isDoubleRing is a private,
 *  non-exported helper, and game modules don't import each other's internals. */
function isDoubleFinish(ring: Throw['ring']): boolean {
  return ring === 'D' || ring === 'IB';
}

interface CopycatVisitHistoryEntry {
  playerId: PlayerId;
  throws: Throw[];
  /** This visit's summed point total. */
  total: number;
  /** What this visit was trying to match. null means it was an unconditional (new-point)
   *  throw with no target to match. */
  targetBefore: number | null;
  /** Did this visit actually score a point (a real match, and — if on match-point — a
   *  qualifying double/inner-bull finish). */
  wasPointWin: boolean;
  /** Which "point" (chase round) this visit belongs to — mirrors X01/Cricket's legNumberBefore,
   *  used by the generic visitHistory accessor to scope the on-screen history to the current
   *  chase round instead of the whole match. */
  pointNumberBefore: number;
  /** Full pre-visit state snapshot, for undo — restored verbatim, never reverse-computed. */
  targetBeforeState: number | null;
  pointsWonBefore: Record<PlayerId, number>;
  currentPlayerIndexBefore: number;
  currentPointNumberBefore: number;
  matchCompleteBefore: boolean;
  winnerIdBefore: PlayerId | null;
}

export interface CopycatState {
  players: PlayerId[];
  currentPlayerIndex: number;
  currentVisit: Throw[];
  /** The total the current visit must match. null only for the very first visit of a new
   *  point — that visit sets a fresh target unconditionally, whatever it is. */
  target: number | null;
  pointsWon: Record<PlayerId, number>;
  /** Starts at 1, increments every time a point is won. */
  currentPointNumber: number;
  matchComplete: boolean;
  winnerId: PlayerId | null;
  visitHistory: CopycatVisitHistoryEntry[];
}

export function init(players: PlayerId[], settings: CopycatSettings): CopycatState {
  const pointsWon: Record<PlayerId, number> = {};
  players.forEach((p) => {
    pointsWon[p] = 0;
  });

  let startPlayerIndex = 0;
  if (settings.firstThrow === 'random') {
    startPlayerIndex = Math.floor(Math.random() * players.length);
  }

  return {
    players,
    currentPlayerIndex: startPlayerIndex,
    currentVisit: [],
    target: null,
    pointsWon,
    currentPointNumber: 1,
    matchComplete: false,
    winnerId: null,
    visitHistory: [],
  };
}

/**
 * Applies one dart to the state and returns a brand-new state object. `state` is never
 * mutated — every nested structure that changes is rebuilt fresh.
 */
export function applyThrow(state: CopycatState, throw_: Throw, settings: CopycatSettings): CopycatState {
  const trustedThrow: Throw = { ...throw_, points: computeDartPoints(throw_.sector, throw_.ring) };
  const newVisit = [...state.currentVisit, trustedThrow];

  if (newVisit.length < 3) {
    return { ...state, currentVisit: newVisit };
  }

  return endVisit(state, newVisit, settings);
}

/** A visit here always resolves at exactly 3 darts — unlike X01, there's no "bust" concept
 *  that could end a visit early (every dart only adds to the total, nothing can go negative),
 *  so there's no early-resolution path to handle. */
function endVisit(state: CopycatState, newVisit: Throw[], settings: CopycatSettings): CopycatState {
  const currentPlayer = state.players[state.currentPlayerIndex];
  const total = newVisit.reduce((sum, t) => sum + t.points, 0);
  const targetBefore = state.target;
  const lastDart = newVisit[newVisit.length - 1];

  const isMatch = targetBefore !== null && total === targetBefore;

  let wasPointWin = false;
  let pointsWonAfter = state.pointsWon;
  let matchCompleteAfter = state.matchComplete;
  let winnerIdAfter = state.winnerId;
  let currentPointNumberAfter = state.currentPointNumber;
  let targetAfter: number | null;
  let nextPlayerIndex: number;

  if (isMatch) {
    const isMatchPoint = state.pointsWon[currentPlayer] === settings.firstTo - 1;
    const qualifies = !isMatchPoint || isDoubleFinish(lastDart.ring);

    if (qualifies) {
      wasPointWin = true;
      pointsWonAfter = { ...state.pointsWon, [currentPlayer]: state.pointsWon[currentPlayer] + 1 };

      if (pointsWonAfter[currentPlayer] >= settings.firstTo) {
        matchCompleteAfter = true;
        winnerIdAfter = currentPlayer;
      } else {
        currentPointNumberAfter = state.currentPointNumber + 1;
      }
      // The point winner leads the next point unconditionally — they don't advance, they
      // throw again next (an unconditional, target-setting visit).
      targetAfter = null;
      nextPlayerIndex = state.currentPlayerIndex;
    } else {
      // Matched, but on match-point without a qualifying finish — doesn't count as a win at
      // all. Treated exactly like a non-match: this total becomes the new target.
      targetAfter = total;
      nextPlayerIndex = (state.currentPlayerIndex + 1) % state.players.length;
    }
  } else {
    // No target yet, or didn't match — this total becomes (or replaces) the target.
    targetAfter = total;
    nextPlayerIndex = (state.currentPlayerIndex + 1) % state.players.length;
  }

  const historyEntry: CopycatVisitHistoryEntry = {
    playerId: currentPlayer,
    throws: newVisit,
    total,
    targetBefore,
    wasPointWin,
    pointNumberBefore: state.currentPointNumber,
    targetBeforeState: state.target,
    pointsWonBefore: state.pointsWon,
    currentPlayerIndexBefore: state.currentPlayerIndex,
    currentPointNumberBefore: state.currentPointNumber,
    matchCompleteBefore: state.matchComplete,
    winnerIdBefore: state.winnerId,
  };

  return {
    ...state,
    currentVisit: [],
    target: targetAfter,
    pointsWon: pointsWonAfter,
    currentPointNumber: currentPointNumberAfter,
    matchComplete: matchCompleteAfter,
    winnerId: winnerIdAfter,
    currentPlayerIndex: matchCompleteAfter ? state.currentPlayerIndex : nextPlayerIndex,
    visitHistory: [...state.visitHistory, historyEntry],
  };
}

export function undo(state: CopycatState): CopycatState {
  if (state.visitHistory.length === 0) return state;

  const last = state.visitHistory[state.visitHistory.length - 1];
  const newHistory = state.visitHistory.slice(0, -1);

  return {
    ...state,
    target: last.targetBeforeState,
    pointsWon: last.pointsWonBefore,
    currentPlayerIndex: last.currentPlayerIndexBefore,
    currentPointNumber: last.currentPointNumberBefore,
    matchComplete: last.matchCompleteBefore,
    winnerId: last.winnerIdBefore,
    currentVisit: [],
    visitHistory: newHistory,
  };
}

export function isComplete(state: CopycatState): boolean {
  return state.matchComplete;
}

export function winner(state: CopycatState): PlayerId | null {
  return state.winnerId;
}
```

```ts
// lib/games/copycat/engine.test.ts
import { describe, it, expect } from 'vitest';
import { init, applyThrow, undo, isComplete, winner } from './engine';
import { createPlayerId } from '@/lib/types';
import { CopycatDefaultSettings } from './schema';

describe('Copycat Engine', () => {
  const player1 = createPlayerId('p1');
  const player2 = createPlayerId('p2');
  const player3 = createPlayerId('p3');

  function throwVisit(sector: number, ring: 'S' | 'D' | 'T' | 'M' | 'OB' | 'IB', times = 1) {
    return Array.from({ length: times }, (_, i) => ({
      sector: sector as any,
      ring,
      points: 0,
      order: i as any,
    }));
  }

  describe('init', () => {
    it('starts every player at 0 points, no target, player 1 first by default', () => {
      const state = init([player1, player2], CopycatDefaultSettings);
      expect(state.pointsWon[player1]).toBe(0);
      expect(state.pointsWon[player2]).toBe(0);
      expect(state.target).toBe(null);
      expect(state.currentPlayerIndex).toBe(0);
      expect(state.currentPointNumber).toBe(1);
    });

    it("'player1', 'bulloff', and 'alternate' all default to index 0 — matching X01's own depth", () => {
      for (const firstThrow of ['player1', 'bulloff', 'alternate'] as const) {
        const state = init([player1, player2], { ...CopycatDefaultSettings, firstThrow });
        expect(state.currentPlayerIndex).toBe(0);
      }
    });
  });

  describe('the first visit of a new point', () => {
    it('sets the target unconditionally, whatever the total', () => {
      let state = init([player1, player2], CopycatDefaultSettings);
      // T20 x3 = 180
      for (const t of throwVisit(20, 'T', 3)) state = applyThrow(state, t, CopycatDefaultSettings);
      expect(state.target).toBe(180);
      expect(state.pointsWon[player1]).toBe(0); // no point — nothing to match yet
      expect(state.players[state.currentPlayerIndex]).toBe(player2); // turn advances normally
    });
  });

  describe('a non-matching visit', () => {
    it('becomes the new target; turn advances normally', () => {
      let state = init([player1, player2], CopycatDefaultSettings);
      for (const t of throwVisit(20, 'T', 3)) state = applyThrow(state, t, CopycatDefaultSettings); // target = 180
      for (const t of throwVisit(5, 'S', 3)) state = applyThrow(state, t, CopycatDefaultSettings); // player2 throws 15, no match
      expect(state.target).toBe(15);
      expect(state.pointsWon[player1]).toBe(0);
      expect(state.pointsWon[player2]).toBe(0);
      expect(state.players[state.currentPlayerIndex]).toBe(player1);
    });
  });

  describe('a matching visit, not on match-point', () => {
    it('scores the point immediately; the winner leads next (does not advance)', () => {
      const settings = { ...CopycatDefaultSettings, firstTo: 5 as const };
      let state = init([player1, player2], settings);
      for (const t of throwVisit(20, 'S', 3)) state = applyThrow(state, t, settings); // target = 60
      for (const t of throwVisit(20, 'S', 3)) state = applyThrow(state, t, settings); // player2 matches 60

      expect(state.pointsWon[player2]).toBe(1);
      expect(isComplete(state)).toBe(false);
      expect(state.target).toBe(null); // reset for the winner's unconditional lead
      expect(state.players[state.currentPlayerIndex]).toBe(player2); // winner leads next, doesn't advance
      expect(state.currentPointNumber).toBe(2);
    });
  });

  describe('match-point', () => {
    it('a matching visit that finishes on a double wins the point and the match', () => {
      const settings = { ...CopycatDefaultSettings, firstTo: 3 as const };
      let state = init([player1, player2], settings);
      state = { ...state, pointsWon: { ...state.pointsWon, [player1]: 2 } }; // player1 on match-point

      for (const t of throwVisit(10, 'S', 3)) state = applyThrow(state, t, settings); // target = 30
      // player2 throws something that doesn't matter for this scenario — advance to player1's turn
      for (const t of throwVisit(1, 'S', 3)) state = applyThrow(state, t, settings); // target = 3, no match, back to player1... 
      // Simpler: reset target directly to exercise player1's match-point attempt in isolation.
      state = { ...state, target: 40, currentPlayerIndex: 0 };
      // player1 throws D20, D20, D... need total 40 with a double last dart: D20 (40) in one dart,
      // then two "safe" non-scoring darts aren't possible (every dart adds to the total) — so
      // finish with the double as dart 3 and low-value darts before it: S1, S1, D19 = 1+1+38 = 40.
      state = applyThrow(state, { sector: 1 as any, ring: 'S', points: 0, order: 0 as any }, settings);
      state = applyThrow(state, { sector: 1 as any, ring: 'S', points: 0, order: 1 as any }, settings);
      state = applyThrow(state, { sector: 19 as any, ring: 'D', points: 0, order: 2 as any }, settings);

      expect(state.pointsWon[player1]).toBe(3);
      expect(isComplete(state)).toBe(true);
      expect(winner(state)).toBe(player1);
    });

    it('a matching visit on match-point WITHOUT a qualifying finish does not win — chase continues', () => {
      const settings = { ...CopycatDefaultSettings, firstTo: 3 as const };
      let state = init([player1, player2], settings);
      state = { ...state, pointsWon: { ...state.pointsWon, [player1]: 2 }, target: 40, currentPlayerIndex: 0 };

      // player1 matches 40 but the last dart is a single, not a double: S20, S19, S1 = 20+19+1 = 40
      state = applyThrow(state, { sector: 20 as any, ring: 'S', points: 0, order: 0 as any }, settings);
      state = applyThrow(state, { sector: 19 as any, ring: 'S', points: 0, order: 1 as any }, settings);
      state = applyThrow(state, { sector: 1 as any, ring: 'S', points: 0, order: 2 as any }, settings);

      expect(state.pointsWon[player1]).toBe(2); // unchanged — did not count
      expect(isComplete(state)).toBe(false);
      expect(state.target).toBe(40); // the (non-qualifying) match total becomes the new target
      expect(state.players[state.currentPlayerIndex]).toBe(player2); // turn advances normally, not a "win"
    });
  });

  describe('undo', () => {
    it('reverts a chase-continuation visit to its exact pre-visit state', () => {
      let state = init([player1, player2], CopycatDefaultSettings);
      for (const t of throwVisit(20, 'T', 3)) state = applyThrow(state, t, CopycatDefaultSettings); // target = 180

      const undone = undo(state);
      expect(undone.target).toBe(null);
      expect(undone.players[undone.currentPlayerIndex]).toBe(player1);
      expect(undone.visitHistory.length).toBe(0);
    });

    it('reverts a point-winning visit, including match completion', () => {
      const settings = { ...CopycatDefaultSettings, firstTo: 1 as const };
      let state = init([player1, player2], settings);
      for (const t of throwVisit(20, 'S', 3)) state = applyThrow(state, t, settings); // target = 60
      for (const t of throwVisit(20, 'S', 3)) state = applyThrow(state, t, settings); // player2 matches, wins (firstTo:1)

      expect(isComplete(state)).toBe(true);

      const undone = undo(state);
      expect(isComplete(undone)).toBe(false);
      expect(undone.pointsWon[player2]).toBe(0);
      expect(undone.target).toBe(60);
      expect(undone.players[undone.currentPlayerIndex]).toBe(player2); // back to player2's matching attempt
    });
  });

  describe('3-player rotation', () => {
    it('after a point is won by a non-first player, they genuinely lead next (not just +1)', () => {
      const settings = { ...CopycatDefaultSettings, firstTo: 5 as const };
      let state = init([player1, player2, player3], settings);
      // player1 leads: target = 60
      for (const t of throwVisit(20, 'S', 3)) state = applyThrow(state, t, settings);
      // player2 doesn't match: target = 15
      for (const t of throwVisit(5, 'S', 3)) state = applyThrow(state, t, settings);
      // player3 matches 15
      for (const t of throwVisit(5, 'S', 3)) state = applyThrow(state, t, settings);

      expect(state.pointsWon[player3]).toBe(1);
      expect(state.players[state.currentPlayerIndex]).toBe(player3); // player3 (index 2) leads next
      expect(state.target).toBe(null);

      // player3's unconditional lead: target = 9
      for (const t of throwVisit(3, 'S', 3)) state = applyThrow(state, t, settings);
      // responder cycle continues in normal order from player3: player1 is next (wraps around)
      expect(state.players[state.currentPlayerIndex]).toBe(player1);
    });
  });
});
```

```ts
// lib/games/copycat/stats.ts
import { Visit, StatTileDef, GameStats } from '@/lib/types';
import { CopycatSettings } from './schema';

export interface CopycatStats extends GameStats {
  points: number;
  avgVisit: number;
  /** Percentage (0-100) of visits thrown against a real target that matched it. null when it
   *  can't be reconstructed — a flat Visit[] log has no `targetBefore` to compare against
   *  (mirrors Cricket's own computeStats returning numbersClosed: null for the same reason). */
  matchRate: number | null;
  /** Points won specifically via a qualifying match-point double/inner-bull finish. null in
   *  computeStats for the same reason as matchRate — a flat Visit[] log has no point-win
   *  context to distinguish an ordinary double hit from a match-point-winning one. */
  doubleFinishes: number | null;
}

export const EMPTY_COPYCAT_STATS: CopycatStats = {
  points: 0,
  avgVisit: 0,
  matchRate: null,
  doubleFinishes: null,
};

export function computeStats(visits: Visit[], settings: CopycatSettings): CopycatStats {
  if (visits.length === 0) {
    return { ...EMPTY_COPYCAT_STATS };
  }

  let totalPoints = 0;
  let pointsWon = 0;

  visits.forEach((visit) => {
    if (visit.isCheckout) pointsWon++;
    visit.throws.forEach((t) => {
      totalPoints += t.points;
    });
  });

  const avgVisit = totalPoints / visits.length;

  return {
    points: pointsWon,
    avgVisit: Math.round(avgVisit * 100) / 100,
    matchRate: null,
    doubleFinishes: null,
  };
}

export const statTiles: StatTileDef<CopycatStats>[] = [
  { label: 'Points', key: 'points', value: (s) => `${s.points}` },
  { label: 'Avg Visit', key: 'avgVisit', value: (s) => s.avgVisit.toFixed(1) },
  {
    label: 'Match %',
    key: 'matchRate',
    value: (s) => (s.matchRate === null ? '—' : `${s.matchRate.toFixed(0)}%`),
  },
  {
    label: 'Double Finishes',
    key: 'doubleFinishes',
    value: (s) => (s.doubleFinishes === null ? '—' : `${s.doubleFinishes}`),
  },
];
```

```ts
// lib/games/copycat/matchStats.ts
import { PlayerId } from '@/lib/types';
import { CopycatState } from './engine';
import { CopycatSettings } from './schema';
import { CopycatStats, EMPTY_COPYCAT_STATS } from './stats';

export function matchStats(state: CopycatState, playerId: PlayerId, settings: CopycatSettings): CopycatStats {
  const entries = state.visitHistory.filter((entry) => entry.playerId === playerId);

  if (entries.length === 0) {
    return { ...EMPTY_COPYCAT_STATS };
  }

  let totalPoints = 0;
  let pointsWon = 0;
  let attemptsAgainstTarget = 0;
  let matches = 0;
  let doubleFinishes = 0;

  entries.forEach((entry) => {
    entry.throws.forEach((t) => {
      totalPoints += t.points;
    });

    if (entry.targetBefore !== null) {
      attemptsAgainstTarget++;
      if (entry.total === entry.targetBefore) matches++;
    }

    if (entry.wasPointWin) {
      pointsWon++;
      const wasMatchPointWin = entry.pointsWonBefore[playerId] === settings.firstTo - 1;
      const lastDart = entry.throws[entry.throws.length - 1];
      const wasDoubleFinish = lastDart.ring === 'D' || lastDart.ring === 'IB';
      if (wasMatchPointWin && wasDoubleFinish) doubleFinishes++;
    }
  });

  const avgVisit = totalPoints / entries.length;
  const matchRate = attemptsAgainstTarget > 0 ? (matches / attemptsAgainstTarget) * 100 : 0;

  return {
    points: pointsWon,
    avgVisit: Math.round(avgVisit * 100) / 100,
    matchRate: Math.round(matchRate * 100) / 100,
    doubleFinishes,
  };
}
```

```ts
// lib/games/copycat/index.ts
import { GameModule, PlayerId, Throw } from '@/lib/types';
import { CopycatState, init as initEngine, applyThrow, undo, isComplete, winner } from './engine';
import { CopycatSettings, CopycatSettingsSchema, CopycatDefaultSettings } from './schema';
import { computeStats, statTiles } from './stats';
import { matchStats } from './matchStats';
import { ScoreboardSlot } from './components';
import { Repeat } from 'lucide-react';

const players = (state: CopycatState): PlayerId[] => state.players;

const currentPlayer = (state: CopycatState): PlayerId => state.players[state.currentPlayerIndex];

const visitProgress = (state: CopycatState): { darts: number; max: number; throws: Throw[] } => ({
  darts: state.currentVisit.length,
  max: 3,
  throws: state.currentVisit,
});

const matchInfo = (state: CopycatState, settings: CopycatSettings): { label: string; value: string }[] => [
  { label: 'Target', value: state.target !== null ? String(state.target) : '—' },
];

const raceProgress = (
  state: CopycatState,
  settings: CopycatSettings
): { playerId: PlayerId; legs: number; legsTarget: number; sets: number; setsTarget: number }[] =>
  state.players.map((playerId) => ({
    playerId,
    legs: state.pointsWon[playerId],
    legsTarget: settings.firstTo,
    sets: 0,
    setsTarget: 0,
  }));

const visitHistory = (
  state: CopycatState,
  playerId: PlayerId
): { throws: Throw[]; scoreBefore: number; scoreAfter: number; busted: boolean }[] =>
  state.visitHistory
    .filter((entry) => entry.playerId === playerId && entry.pointNumberBefore === state.currentPointNumber)
    .map((entry) => ({
      throws: entry.throws,
      scoreBefore: 0,
      scoreAfter: entry.total,
      busted: false,
    }))
    .reverse();

const heatmap = (state: CopycatState, playerId: PlayerId): Record<string, number> => {
  const counts: Record<string, number> = {};
  state.visitHistory
    .filter((entry) => entry.playerId === playerId)
    .forEach((entry) => {
      entry.throws.forEach((t) => {
        const key = `${t.ring}${t.sector}`;
        counts[key] = (counts[key] ?? 0) + 1;
      });
    });
  return counts;
};

const lastCompletedVisit = (
  state: CopycatState,
  playerId: PlayerId
): { throws: Throw[]; scoreBefore: number; scoreAfter: number; busted: boolean; isCheckout: boolean } | null => {
  const entries = state.visitHistory.filter((entry) => entry.playerId === playerId);
  if (entries.length === 0) return null;
  const latest = entries[entries.length - 1];
  return {
    throws: latest.throws,
    scoreBefore: 0,
    scoreAfter: latest.total,
    busted: false,
    isCheckout: latest.wasPointWin,
  };
};

const CopycatModule: GameModule<CopycatState, CopycatSettings> = {
  id: 'copycat',
  name: 'Copycat',
  description: 'Match the total to score — win the point and lead next',
  icon: Repeat,
  playerRange: [2, 8],
  status: 'stable',

  settingsSchema: CopycatSettingsSchema,
  defaultSettings: CopycatDefaultSettings,

  init: initEngine,
  applyThrow,
  undo,
  isComplete,
  winner,

  players,
  currentPlayer,
  visitProgress,
  matchInfo,
  raceProgress,
  visitHistory,
  heatmap,
  lastCompletedVisit,
  matchStats,

  ScoreboardSlot,
  primaryMetric: (state: CopycatState, playerId: PlayerId) => ({
    label: 'Points',
    value: state.pointsWon[playerId].toString(),
  }),

  computeStats,
  statTiles,
};

export default CopycatModule;
```

Note: `scoreBefore: 0` (not `entry.targetBefore ?? 0`) in `visitHistory`/`lastCompletedVisit` is deliberate — the generic history UI displays `Math.abs(scoreAfter - scoreBefore)` per visit, and the only thing worth showing there is the visit's own total, not a target-vs-total difference. Setting `scoreBefore: 0` makes that delta exactly equal `entry.total`.

- [ ] **Step 2: Write index.test.ts — the lastCompletedVisit boundary case**

X01 has an established precedent for a real bug class here (see `lib/games/x01/index.test.ts`): a visit that wins a point advances `currentPointNumber` as part of the *same* transition that recorded it — so the generic `visitHistory` accessor (filtered to `pointNumberBefore === state.currentPointNumber`) excludes that exact visit immediately afterward, the moment it happens. `lastCompletedVisit` exists precisely to still find it. This needs its own test, not just engine-level coverage, because it's a module-level (`GameModule` interface) concern.

```ts
// lib/games/copycat/index.test.ts
import { describe, it, expect } from 'vitest';
import CopycatModule from './index';
import { createPlayerId } from '@/lib/types';
import { CopycatDefaultSettings } from './schema';

describe('CopycatModule.lastCompletedVisit', () => {
  const player1 = createPlayerId('p1');
  const player2 = createPlayerId('p2');

  it('returns null when the player has no completed visits yet', () => {
    const state = CopycatModule.init([player1, player2], CopycatDefaultSettings);
    expect(CopycatModule.lastCompletedVisit(state, player1)).toBe(null);
  });

  it('still returns a visit that won a point, even though visitHistory no longer does', () => {
    const settings = { ...CopycatDefaultSettings, firstTo: 5 as const };
    let state = CopycatModule.init([player1, player2], settings);
    const throwSector = (sector: number, ring: 'S' | 'D' | 'T') => ({
      sector: sector as any,
      ring,
      points: 0,
      order: 0 as any,
    });

    // player1: S20 x3 = 60 (unconditional, sets target)
    state = CopycatModule.applyThrow(state, throwSector(20, 'S'), settings);
    state = CopycatModule.applyThrow(state, throwSector(20, 'S'), settings);
    state = CopycatModule.applyThrow(state, throwSector(20, 'S'), settings);
    // player2: S20 x3 = 60 (matches, wins the point)
    state = CopycatModule.applyThrow(state, throwSector(20, 'S'), settings);
    state = CopycatModule.applyThrow(state, throwSector(20, 'S'), settings);
    state = CopycatModule.applyThrow(state, throwSector(20, 'S'), settings);

    expect(state.currentPointNumber).toBe(2); // advanced as part of this same throw
    expect(CopycatModule.visitHistory(state, player2)).toHaveLength(0); // filtered out — the bug this accessor fixes

    const visit = CopycatModule.lastCompletedVisit(state, player2);
    expect(visit).not.toBe(null);
    expect(visit!.scoreAfter).toBe(60);
    expect(visit!.isCheckout).toBe(true);
  });
});
```
